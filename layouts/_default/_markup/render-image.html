{{- /* Image render hook: for images stored next to a post (page bundle), Hugo makes web-sized copies (800 / 1200 / 1600px wide) and the browser picks the right one. The original upload is never published, so you can upload full-size files, and camera metadata (including GPS) is stripped. Anything else (remote URLs, SVG, GIF) is passed through unchanged. */ -}} {{- $u := urls.Parse .Destination -}} {{- $alt := .Text -}} {{- $title := .Title -}} {{- $img := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $img = or (.PageInner.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- if and $img (eq $img.ResourceType "image") (in (slice "jpeg" "png" "webp" "tiff" "bmp") $img.MediaType.SubType) -}} {{- $orig := $img -}} {{- /* Rotate phone photos the right way up before resizing */ -}} {{- if eq $img.MediaType.SubType "jpeg" }}{{ $orig = $img.Filter images.AutoOrient }}{{ end -}} {{- $widths := slice -}} {{- range slice 800 1200 1600 -}} {{- if lt . $orig.Width }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- if lt (len $widths) 3 }}{{ $widths = $widths | append (math.Min $orig.Width 1600 | int) }}{{ end -}} {{- $srcset := slice -}} {{- $largest := "" -}} {{- range $widths -}} {{- $r := $orig.Resize (printf "%dx jpg q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- $largest = $r -}} {{- end -}} {{- else -}} {{- $src := $u.String -}} {{- with $img }}{{ $src = .RelPermalink }}{{ end -}} {{- end -}}
